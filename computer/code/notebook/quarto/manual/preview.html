import plotly.graph_objects as go
import numpy as np

# 创建数据
x = np.linspace(0, 10, 100)
y = np.sin(x)

# 创建图表
fig = go.Figure()

# 添加数据
fig.add_trace(go.Scatter(x=x, y=y, mode='lines', name='Sine Wave'))

# 设置标题和坐标轴标签
fig.update_layout(title="交互式正弦曲线", xaxis_title="x", yaxis_title="y")

# 显示图表
fig.show()
